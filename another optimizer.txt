-- ========================================================================
-- 0. Black Loading Screen (shows immediately, fades out once fully loaded)
-- ========================================================================
local blackScreenGui

local function showBlackLoadingScreen()
    local localPlayer = Players.LocalPlayer
    if not localPlayer then return end

    local playerGui = localPlayer:WaitForChild("PlayerGui", 5)
    if not playerGui then return end

    local existing = playerGui:FindFirstChild("SxlvatoreBlackScreenGui")
    if existing then existing:Destroy() end

    blackScreenGui = Instance.new("ScreenGui")
    blackScreenGui.Name = "SxlvatoreBlackScreenGui"
    blackScreenGui.ResetOnSpawn = false
    blackScreenGui.IgnoreGuiInset = true
    blackScreenGui.DisplayOrder = 999998

    local blackFrame = Instance.new("Frame")
    blackFrame.Name = "BlackFrame"
    blackFrame.Size = UDim2.new(1, 0, 1, 0)
    blackFrame.Position = UDim2.new(0, 0, 0, 0)
    blackFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
    blackFrame.BackgroundTransparency = 0
    blackFrame.BorderSizePixel = 0
    blackFrame.Parent = blackScreenGui

    local loadingLabel = Instance.new("TextLabel")
    loadingLabel.Name = "LoadingLabel"
    loadingLabel.Size = UDim2.new(1, 0, 0, 40)
    loadingLabel.Position = UDim2.new(0, 0, 1, -80)
    loadingLabel.BackgroundTransparency = 1
    loadingLabel.Text = "Loading..."
    loadingLabel.TextColor3 = Color3.fromRGB(180, 180, 180)
    loadingLabel.Font = Enum.Font.Gotham
    loadingLabel.TextSize = 14
    loadingLabel.Parent = blackFrame

    blackScreenGui.Parent = playerGui
end

local function hideBlackLoadingScreen()
    if not blackScreenGui then return end
    local blackFrame = blackScreenGui:FindFirstChild("BlackFrame")
    if blackFrame then
        local fadeInfo = TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        local fadeTween = TweenService:Create(blackFrame, fadeInfo, {BackgroundTransparency = 1})
        local labelTween
        local loadingLabel = blackFrame:FindFirstChild("LoadingLabel")
        if loadingLabel then
            labelTween = TweenService:Create(loadingLabel, fadeInfo, {TextTransparency = 1})
            labelTween:Play()
        end
        fadeTween:Play()
        fadeTween.Completed:Connect(function()
            blackScreenGui:Destroy()
            blackScreenGui = nil
        end)
    else
        blackScreenGui:Destroy()
        blackScreenGui = nil
    end
end

-- 2. Sxlvatore Notifier (now includes rainbow load-time readout)
local function showTopNotifier(loadTimeSeconds)
    local localPlayer = Players.LocalPlayer
    if not localPlayer then return end

    local playerGui = localPlayer:WaitForChild("PlayerGui", 5)
    if not playerGui then return end

    local existing = playerGui:FindFirstChild("SxlvatoreNotifierGui")

    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "SxlvatoreGui"
    screenGui.ResetOnSpawn = false
    screenGui.IgnoreGuiInset = true
    screenGui.DisplayOrder = 999999

    local mainFrame = Instance.new("Frame")
    mainFrame.Name = "MainFrame"

    mainFrame.Position = UDim2.new(0.5, -170, 0, -80)

    mainFrame.Parent = screenGui

    local uiCorner = Instance.new("UICorner")
    uiCorner.CornerRadius = UDim.new(0, 10)
    uiCorner.Parent = mainFrame

    local uiStroke = Instance.new("UIStroke")
    uiStroke.Color = Color3.fromRGB(120, 80, 255)
    uiStroke.Thickness = 1.5
    uiStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    uiStroke.Parent = mainFrame

    local textLabel = Instance.new("TextLabel")
    textLabel.Name = "NotifyTitle"
    textLabel.Size = UDim2.new(1, 0, 0, 22)
    textLabel.Position = UDim2.new(0, 0, 0, 6)
    textLabel.BackgroundTransparency = 1
    textLabel.Text = "Sxlvatore Optimizer"
    textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    textLabel.Font = Enum.Font.GothamBold
    textLabel.TextSize = 16
    textLabel.Parent = mainFrame

    local subLabel = Instance.new("TextLabel")
    subLabel.Name = "NotifySubtitle"
    subLabel.Size = UDim2.new(1, 0, 0, 14)
    subLabel.Position = UDim2.new(0, 0, 0, 28)
    subLabel.BackgroundTransparency = 1
    subLabel.Text = "OPTIMIZATIONS & FFLAGS LOADED"
    subLabel.TextColor3 = Color3.fromRGB(0, 255, 180)
    subLabel.Font = Enum.Font.GothamBold
    subLabel.TextSize = 10
    subLabel.Parent = mainFrame

    -- Rainbow load-time label, sits right under the subtitle
    local loadTimeLabel = Instance.new("TextLabel")
    loadTimeLabel.Name = "LoadTimeLabel"
    loadTimeLabel.Size = UDim2.new(1, 0, 0, 16)
    loadTimeLabel.Position = UDim2.new(0, 0, 0, 44)
    loadTimeLabel.BackgroundTransparency = 1
    loadTimeLabel.Text = string.format("Load Time: %.2fs", loadTimeSeconds or 0)
    loadTimeLabel.TextColor3 = Color3.fromRGB(255, 0, 0) -- overwritten every frame below
    loadTimeLabel.Font = Enum.Font.GothamBold
    loadTimeLabel.TextSize = 12
    loadTimeLabel.Parent = mainFrame

    screenGui.Parent = playerGui

    -- Rainbow color cycle on the load time text
    local hue = 0
    local rainbowConn
    rainbowConn = RunService.Heartbeat:Connect(function(dt)
        hue = (hue + dt * 0.5) % 1 -- full cycle every ~2 seconds
        loadTimeLabel.TextColor3 = Color3.fromHSV(hue, 1, 1)
    end)

    local slideInInfo = TweenInfo.new(0.5, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    local slideOutInfo = TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.In)

    local tweenIn = TweenService:Create(mainFrame, slideInInfo, {Position = UDim2.new(0.5, -170, 0, 20)})
    local tweenOut = TweenService:Create(mainFrame, slideOutInfo, {Position = UDim2.new(0.5, -170, 0, -80)})

    tweenIn:Play()

    task.delay(4, function()
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            if rainbowConn then
                rainbowConn:Disconnect()
                rainbowConn = nil
            end
            screenGui:Destroy()
        end)
    end)
end

-- ========================================================================
-- Optimized Loading: Apply FFlags BEFORE waiting for game to load
-- so Roblox can use them as early as possible
-- ========================================================================

local function waitUntilFullyLoadedThenHide()
    if not game:IsLoaded() then
        game.Loaded:Wait()
    end

    local localPlayer = Players.LocalPlayer
    if localPlayer then
        if not localPlayer.Character then
            localPlayer.CharacterAdded:Wait()
        end
    end

    -- Reduced wait: 0.1s instead of 0.5s for faster reveal
    task.wait(0.1)

    -- Stop the clock right before we reveal the game
    measuredLoadTime = os.clock() - scriptStartTime

    hideBlackLoadingScreen()
    showTopNotifier(measuredLoadTime)
end

task.spawn(showBlackLoadingScreen)

-- ========================================================================
-- 3. Actually apply the FFlags
-- ========================================================================
local function getFFlagSetter()
    local candidates = {
        "setfflag",
        "setfastflag",
        "set_fflag",
        "setFFlag",
        "fflag",
        "mfflags",
    }

    for _, name in ipairs(candidates) do
        local fn = rawget(getfenv(), name) or (getgenv and getgenv()[name])
        if type(fn) == "function" then
            return fn
        end
    end
    return nil
end

local function applyFFlags()
    local setter = getFFlagSetter()

    if not setter then
        warn("[Sxlvatore Optimizer] No FFlag-setting function found in this environment. "
            .. "FFlags were NOT applied. Check your executor's documentation for the "
            .. "correct global function name (commonly setfflag/setfastflag).")
        return 0
    end

    local applied, failed = 0, 0
    for flagName, flagValue in pairs(fflags) do
        local ok = pcall(function()
            setter(flagName, flagValue)
        end)
        if ok then
            applied += 1
        else
            failed += 1
        end
    end

    print(string.format("[Sxlvatore Optimizer] FFlags applied: %d, failed: %d", applied, failed))
    return applied
end

-- Apply FFlags first (before loading wait) for maximum effect
task.spawn(applyFFlags)
-- Then wait for the game to be loaded and hide the screen
task.spawn(waitUntilFullyLoadedThenHide)
