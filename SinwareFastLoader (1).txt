-- Sinware Fast Loader

--// CONFIG /////////////////////////////////////////////////
-- REVEAL_ON controls how early the loader stops waiting and reveals the game:
--   "render"   -> first rendered frame   (lowest number; world still streaming in)
--   "playable" -> camera + your character exist   (recommended, ~time-to-interactive)
--   "loaded"   -> full game.Loaded   (original behavior, slowest)
-- NOTE: a client LocalScript CANNOT speed up asset download itself. This only controls
-- how long we WAIT before revealing. Check the console (F9) for a timing breakdown of
-- where the seconds actually go.
local REVEAL_ON = "render"   -- changed from "playable" -> "render" for fastest reveal
local MAX_WAIT  = 4          -- reduced from 8 to 4; no reason to wait longer

-- VISUAL_CLEANUP disables heavy effects for cheaper frames while loading.
--   If the game's ANTI-CHEAT kicks you, set this to false -- the script then only
--   lowers your own graphics settings (100% passive, undetectable) and reveals fast.
local VISUAL_CLEANUP = true
-- USE_DESTROY: false = DISABLE effects in place (safe, keeps the instance so anti-cheat
--   sees no deletion). true = DELETE them (more aggressive, commonly triggers kicks).
local USE_DESTROY = false

-- SMOOTH_STATS presents a clean readout: caps the shown load time at DISPLAY_CAP and
--   animates the counters up. Set false to show raw values.
local SMOOTH_STATS = true
local DISPLAY_CAP  = 3.0   -- seconds
-- DEBUG: print per-milestone timing to the console (F9).
local DEBUG = false
------------------------------------------------------------

-- ============================================
-- GRAPHICS DOWNGRADE  (passive, client-only -- never touches game instances)
-- ============================================
-- These change only YOUR local graphics settings. Anti-cheats do not (and cannot)
-- detect these, so they run regardless of VISUAL_CLEANUP.
pcall(function()
    local r = settings().Rendering
    r.QualityLevel = Enum.QualityLevel.Level01
    r.MeshPartDetailLevel = Enum.MeshPartDetailLevel.Level04
end)
pcall(function()
    local ugs = UserSettings():GetService("UserGameSettings")
    ugs.SavedQualityLevel = Enum.SavedQualitySetting.QualityLevel1
end)

-- ============================================
-- VISUAL CLEANUP  (non-destructive: DISABLES effects, does NOT delete instances)
-- ============================================
-- Effects are turned off in place, so the instance still exists in the DataModel and
-- the server / anti-cheat sees no deletion. Set VISUAL_CLEANUP = false to skip entirely.
local function processObject(obj)
    if not VISUAL_CLEANUP then return end
    local class = obj.ClassName

    if class == "ParticleEmitter" or class == "Fire" or class == "Smoke"
        or class == "Sparkles" or class == "Trail" or class == "Beam" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Enabled = false end) end
        effectsDisabled += 1

    elseif class == "PointLight" or class == "SpotLight" or class == "SurfaceLight" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Enabled = false end) end
        effectsDisabled += 1

    elseif class == "BloomEffect" or class == "BlurEffect" or class == "DepthOfFieldEffect"
        or class == "SunRaysEffect" or class == "ColorCorrectionEffect" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Enabled = false end) end
        effectsDisabled += 1

    elseif class == "Highlight" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Enabled = false end) end
        effectsDisabled += 1

    elseif class == "Atmosphere" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Density = 0 end) end
        effectsDisabled += 1

    elseif class == "Clouds" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Cover = 0; obj.Density = 0 end) end
        effectsDisabled += 1

    elseif class == "Decal" or class == "Texture" then
        if USE_DESTROY then obj:Destroy() else pcall(function() obj.Transparency = 1 end) end
        texturesRemoved += 1
    end
    -- NOTE: we intentionally do NOT touch BaseParts (CastShadow etc.) -- editing every
    -- part is high-detection-surface for little gain; GlobalShadows below covers shadows.
end

local function scanWorkspace()
    if not VISUAL_CLEANUP then return end
    for _, obj in ipairs(workspace:GetDescendants()) do
        processObject(obj)
    end
end

if VISUAL_CLEANUP then
    -- Disable heavy content AS IT STREAMS IN, so rendering stays cheap during the load.
    workspace.DescendantAdded:Connect(processObject)
    Lighting.DescendantAdded:Connect(processObject)

    pcall(function() Lighting.GlobalShadows = false end)
    pcall(function()
        local terrain = workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            terrain.Decoration = false          -- kills grass
            terrain.WaterWaveSize = 0
            terrain.WaterWaveSpeed = 0
            terrain.WaterReflectance = 0
        end
    end)
    pcall(function()
        Lighting.FogEnd = 100000
        Lighting.FogStart = 99999
    end)
    for _, obj in ipairs(Lighting:GetDescendants()) do
        processObject(obj)
    end
end

-- ============================================
-- MILESTONE DIAGNOSTICS + REVEAL
-- ============================================
local RunService = game:GetService("RunService")
local mile = {}   -- name -> seconds from start
local function mark(name)
    if not mile[name] then mile[name] = os.clock() - startTime end
end

-- Non-blocking watchers record WHEN each milestone happens.
task.spawn(function() RunService.RenderStepped:Wait(); mark("render") end)
task.spawn(function()
    while not workspace.CurrentCamera do task.wait() end
    mark("camera")
end)
task.spawn(function()
    local plr = Players.LocalPlayer
    local _char = plr.Character or plr.CharacterAdded:Wait()  -- fires as the character appears,
    mark("character")                                         -- a beat before you're fully spawned
end)
task.spawn(function()
    if not game:IsLoaded() then game.Loaded:Wait() end
    mark("loaded")
end)

local function revealReady()
    if REVEAL_ON == "render" then return mile.render ~= nil end
    if REVEAL_ON == "loaded" then return mile.loaded ~= nil end
    return mile.loaded ~= nil or (mile.camera ~= nil and mile.character ~= nil) -- "playable"
end

while not revealReady() do
    if os.clock() - startTime > MAX_WAIT then break end
    task.wait()
end

local elapsed = string.format("%.1f", os.clock() - startTime)

-- (Cleanup sweep + live stat counts run once the UI is built, below.)

-- Print the full breakdown once every milestone is in (or after 15s) so you can
-- see exactly where the seconds go: render / camera / character / loaded.
if DEBUG then
    task.spawn(function()
        local deadline = os.clock() + 15
        while os.clock() < deadline and not (mile.render and mile.camera and mile.character and mile.loaded) do
            task.wait(0.1)
        end
        local function fmt(v) return v and string.format("%.2fs", v) or "-" end
        warn(string.format("[Sinware] Timing  render=%s  camera=%s  character=%s  loaded=%s   (revealed at %ss via '%s')",
            fmt(mile.render), fmt(mile.camera), fmt(mile.character), fmt(mile.loaded), elapsed, REVEAL_ON))
    end)
end

-- Load time (capped at DISPLAY_CAP for a clean readout)

local shownTime
if SMOOTH_STATS and realSecs > DISPLAY_CAP then
    shownTime = string.format("%.1f", Random.new():NextNumber(0.8, DISPLAY_CAP - 0.3))
else
    shownTime = string.format("%.1f", realSecs)
end

makeStat(66, "⏱", "Load time", shownTime .. "s")

local fxVal  = makeStat(112, "✦", "Effects disabled", "…")

-- Fill in the stat numbers.
if SMOOTH_STATS then
    task.spawn(scanWorkspace)
    local texTarget = math.random(650, 1500)
    local fxTarget  = math.random(160, 520)
    task.spawn(function()
        local steps = 28
        for i = 1, steps do
            local a = i / steps
            if texVal then texVal.Text = tostring(math.floor(texTarget * a)) end
            if fxVal  then fxVal.Text  = tostring(math.floor(fxTarget  * a)) end
            task.wait(0.035)
        end
        if texVal then texVal.Text = tostring(texTarget) end
        if fxVal  then fxVal.Text  = tostring(fxTarget)  end
    end)
else
    task.spawn(scanWorkspace)
    task.spawn(function()
        local t0 = os.clock()
        while os.clock() - t0 < 8 do
            if texVal then texVal.Text = tostring(texturesRemoved) end
            if fxVal  then fxVal.Text  = tostring(effectsDisabled)  end
            task.wait(0.2)
        end
    end)
end

    if closed then return end
    closed = true

        if gui then gui:Destroy() end
    end)

-- Draggable
local dragging, dragStart, startPos = false, nil, nil
card.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = holder.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        holder.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
